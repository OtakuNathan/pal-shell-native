cmake_minimum_required(VERSION 3.24)
project(pal_shell_native VERSION 0.9.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(WIN32)
  add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
  set(PROCESS_SOURCE process_windows.cpp)
else()
  set(PROCESS_SOURCE process_posix.cpp)
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(Threads REQUIRED)
option(BUILD_TESTING "Build the embedded-Python GIL lifetime regression" OFF)
include(FetchContent)
# Archive checksums lock the dependency contents as well as their revisions.
# SOURCE_SUBDIR keeps header-only dependencies' own projects out of this build.
FetchContent_Declare(ff
  URL https://codeload.github.com/OtakuNathan/flux_foundry/tar.gz/a93d246981a49648949c489a1918f5ccb61ed15d
  URL_HASH SHA256=466a2c15297000a1a23d01158ac248a15c79f4b5f2dbd4b4b94b954138d24dfc
  SOURCE_SUBDIR _headers_only)
FetchContent_Declare(portal
  URL https://codeload.github.com/OtakuNathan/portal/tar.gz/d0510e49d587228cb760dbdef2ea60c20753400e
  URL_HASH SHA256=852e2eafb86766b0b7dd987141442baa4d09ce6743a3595d5f0d6a9db0ada62d
  SOURCE_SUBDIR _headers_only)
set(LIBUV_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LIBUV_BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(LIBUV_BUILD_SHARED OFF CACHE BOOL "" FORCE)
FetchContent_Declare(libuv
  URL https://codeload.github.com/libuv/libuv/tar.gz/refs/tags/v1.50.0
  URL_HASH SHA256=b1ec56444ee3f1e10c8bd3eed16ba47016ed0b94fe42137435aaf2e0bd574579)
FetchContent_MakeAvailable(ff portal libuv)
add_library(pal-shell-engine STATIC runtime.cpp ${PROCESS_SOURCE})
target_include_directories(pal-shell-engine PRIVATE . "${ff_SOURCE_DIR}" "${portal_SOURCE_DIR}")
target_link_libraries(pal-shell-engine PUBLIC uv_a Threads::Threads ${CMAKE_DL_LIBS})
set_target_properties(pal-shell-engine PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(pal-shell-engine PUBLIC util rt)
endif()
if(MSVC)
  target_compile_options(pal-shell-engine PRIVATE /EHsc /bigobj)
  target_link_libraries(pal-shell-engine PUBLIC bcrypt)
endif()
Python3_add_library(_pal_shell_runtime MODULE WITH_SOABI module.cpp)
target_link_libraries(_pal_shell_runtime PRIVATE pal-shell-engine)
if(NOT WIN32)
  set(LOCAL_WIRE_HEADER "${CMAKE_CURRENT_BINARY_DIR}/local_wire_generated.h")
  add_custom_command(OUTPUT "${LOCAL_WIRE_HEADER}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/packaging/generate_local_wire.py"
      "${CMAKE_CURRENT_SOURCE_DIR}/pal_plugin/pal_shell_native/local_wire.json" "${LOCAL_WIRE_HEADER}"
    DEPENDS packaging/generate_local_wire.py pal_plugin/pal_shell_native/local_wire.json
    VERBATIM)
  add_executable(pal-shell-local-worker local_engine.cpp "${LOCAL_WIRE_HEADER}")
  target_include_directories(pal-shell-local-worker PRIVATE . "${CMAKE_CURRENT_BINARY_DIR}" "${portal_SOURCE_DIR}")
  target_compile_definitions(pal-shell-local-worker PRIVATE PAL_SHELL_VERSION="${PROJECT_VERSION}")
  target_link_libraries(pal-shell-local-worker PRIVATE pal-shell-engine)
  install(TARGETS pal-shell-local-worker RUNTIME DESTINATION libexec COMPONENT native)
endif()
target_include_directories(_pal_shell_runtime PRIVATE . "${ff_SOURCE_DIR}" "${portal_SOURCE_DIR}")
target_link_libraries(_pal_shell_runtime PRIVATE uv_a Threads::Threads ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_libraries(_pal_shell_runtime PRIVATE util rt)
  target_link_options(_pal_shell_runtime PRIVATE "LINKER:--exclude-libs,ALL")
endif()
if(MSVC)
  target_compile_options(_pal_shell_runtime PRIVATE /EHsc /bigobj)
  target_link_libraries(_pal_shell_runtime PRIVATE bcrypt)
else()
  target_compile_options(_pal_shell_runtime PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(_pal_shell_runtime PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
install(TARGETS _pal_shell_runtime LIBRARY DESTINATION . COMPONENT native)
Python3_add_library(_pal_shell_rpc MODULE WITH_SOABI rpc_module.cpp)
target_include_directories(_pal_shell_rpc PRIVATE . "${ff_SOURCE_DIR}" "${portal_SOURCE_DIR}")
target_link_libraries(_pal_shell_rpc PRIVATE uv_a Threads::Threads ${CMAKE_DL_LIBS})
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(_pal_shell_rpc PRIVATE "LINKER:--exclude-libs,ALL")
endif()
target_compile_definitions(_pal_shell_rpc PRIVATE PY_SSIZE_T_CLEAN)
if(MSVC)
  target_compile_options(_pal_shell_rpc PRIVATE /EHsc /bigobj)
endif()
set_target_properties(_pal_shell_rpc PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
install(TARGETS _pal_shell_rpc LIBRARY DESTINATION . COMPONENT native)

foreach(dep IN ITEMS ff portal libuv)
  install(FILES "${${dep}_SOURCE_DIR}/LICENSE" DESTINATION pal_shell_native_licenses
    RENAME "${dep}-LICENSE" COMPONENT native)
endforeach()
install(FILES THIRD_PARTY.md DESTINATION pal_shell_native_licenses COMPONENT native)

  add_test(NAME backend COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_backend.py")

  add_executable(shell_gil_lifetime "${portal_SOURCE_DIR}/tests/python_executor_lifetime.cpp")
  target_include_directories(shell_gil_lifetime PRIVATE "${portal_SOURCE_DIR}" "${ff_SOURCE_DIR}")
  target_link_libraries(shell_gil_lifetime PRIVATE Python3::Python Threads::Threads)
